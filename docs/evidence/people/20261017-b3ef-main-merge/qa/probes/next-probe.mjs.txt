import { createRequire } from "node:module";
import { readFileSync, writeFileSync } from "node:fs";
import { createHash } from "node:crypto";
const repo = "/Users/leegwanghui/dev/hackathon/sol-radar";
const evidence =
  "/Users/leegwanghui/dev/hackathon/sol-radar-inputs/main-merges/20261017-df51f2f/qa";
const require = createRequire(repo + "/web/package.json");
const { chromium } = require("@playwright/test");
const manifest = JSON.parse(
  readFileSync(repo + "/fixtures/people/solideo.portraits.json", "utf8"),
).portraits;
const origin = "http://localhost:3862";
const gets = [];
let offset = 0;
await Promise.all(
  Array.from({ length: 8 }, async () => {
    while (offset < manifest.length) {
      const item = manifest[offset++];
      const response = await fetch(origin + item.path);
      const body = Buffer.from(await response.arrayBuffer());
      const result = {
        personId: item.personId,
        path: item.path,
        status: response.status,
        mime: response.headers.get("content-type"),
        bytes: body.length,
        sha256: createHash("sha256").update(body).digest("hex"),
      };
      result.pass =
        result.status === 200 &&
        result.mime?.startsWith("image/png") &&
        result.sha256 === item.sha256 &&
        result.bytes === readFileSync(repo + "/web/public" + item.path).length;
      gets.push(result);
    }
  }),
);
gets.sort((a, b) => a.personId.localeCompare(b.personId));
writeFileSync(
  evidence + "/next-200-get.json",
  JSON.stringify(gets, null, 2) + "\n",
);
if (gets.length !== 200 || gets.some((x) => !x.pass))
  throw Error("Next GET contract failed; see evidence");
const browser = await chromium.launch();
try {
  const page = await browser.newPage({
    viewport: { width: 1440, height: 1000 },
    locale: "ko-KR",
    timezoneId: "Asia/Seoul",
  });
  const errors = [];
  page.on("pageerror", (e) => errors.push(e.message));
  await page.goto(origin + "/auth/sign-up");
  await page.getByRole("heading", { name: "회원가입", exact: true }).waitFor();
  const decoded = await page.evaluate(
    async (paths) => {
      let index = 0;
      const rows = [];
      await Promise.all(
        Array.from({ length: 4 }, async () => {
          while (index < paths.length) {
            const path = paths[index++];
            const img = new Image();
            img.src = path;
            try {
              await img.decode();
              rows.push({
                path,
                width: img.naturalWidth,
                height: img.naturalHeight,
                pass: img.naturalWidth === 1254 && img.naturalHeight === 1254,
              });
            } catch {
              rows.push({ path, pass: false });
            }
          }
        }),
      );
      return rows;
    },
    manifest.map((p) => p.path),
  );
  writeFileSync(
    evidence + "/next-200-browser-decode.json",
    JSON.stringify(
      decoded.sort((a, b) => a.path.localeCompare(b.path)),
      null,
      2,
    ) + "\n",
  );
  if (decoded.length !== 200 || decoded.some((x) => !x.pass))
    throw Error("Browser decoding failed");
  const controls = [];
  for (const width of [1440, 1280]) {
    await page.setViewportSize({ width, height: 1000 });
    await page.goto(origin + "/auth/sign-up");
    await page.getByRole("combobox", { name: "부서 (필수)" }).click();
    await page
      .getByRole("option", { name: "AX사업본부 AI플랫폼운영팀", exact: true })
      .click();
    await page.getByRole("combobox", { name: "직급/직책 (필수)" }).click();
    await page.getByRole("option", { name: "팀장", exact: true }).click();
    const overflow = await page.evaluate(() =>
      Math.max(0, document.documentElement.scrollWidth - window.innerWidth),
    );
    controls.push({
      width,
      department: await page
        .getByRole("combobox", { name: "부서 (필수)" })
        .innerText(),
      title: await page
        .getByRole("combobox", { name: "직급/직책 (필수)" })
        .innerText(),
      overflow,
    });
    await page
      .locator('[data-slot="select-content"]')
      .waitFor({ state: "detached" });
    await page.screenshot({
      path: evidence + "/public-signup-" + width + ".png",
      fullPage: true,
    });
  }
  writeFileSync(
    evidence + "/next-browser-ui.json",
    JSON.stringify({ controls, pageErrors: errors }, null, 2) + "\n",
  );
  if (controls.some((x) => x.overflow !== 0) || errors.length)
    throw Error("Public UI probe failed");
  console.log(
    JSON.stringify({
      nextGETPass: gets.length,
      browserDecoded: decoded.length,
      desktopControls: controls,
      pageErrors: errors.length,
    }),
  );
} finally {
  await browser.close();
}
